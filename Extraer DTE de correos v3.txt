/**
 * ====================================================================================
 * SCRIPT DE EXTRACCIÓN Y ORGANIZACIÓN DE DTE DESDE GMAIL (GOOGLE APPS SCRIPT) - v3
 * ====================================================================================
 *
 * ¿QUÉ HACE ESTE SCRIPT?
 * Revisa la bandeja de entrada de Gmail, encuentra los Documentos Tributarios Electrónicos (DTE
 * de El Salvador) que llegan por correo, los guarda ordenados en Google Drive y deja un registro
 * de cada archivo en una hoja de cálculo por empresa.
 *
 * FLUJO GENERAL (función de entrada: processDTEAttachments, pensada para un trigger por tiempo):
 *  1. Busca hilos nuevos: "in:inbox has:nouserlabels" (los ya procesados quedan con etiqueta).
 *  2. Por cada mensaje obtiene los adjuntos; si hay .zip los descomprime en memoria y toma los
 *     .json / .pdf / .xml 
 *  3. Si hay JSON: lo lee (reparándolo si viene roto), identifica la empresa por NIT (emisor =
 *     Ventas, receptor = Compras) y busca su PDF/XML pareja por nombre o por enlace externo.
 *  4. Si NO hay JSON adjunto: sondea los enlaces del correo (portales tipo Walmart/Edicom) para
 *     descargar JSON/PDF/ZIP; si el portal está protegido usa los datos del texto del correo.
 *  5. Valida SIEMPRE la pareja PDF + JSON: si falta uno deja un registro "PDF - No encontrado" o
 *     "JSON - No encontrado" indicando de qué archivo recibido es contraparte (y lo anota en el log).
 *  6. ANTI-DUPLICADOS: antes de guardar, verifica por Código de Generación + formato si ese archivo
 *     ya fue registrado; si ya existe lo omite (no lo sube a Drive ni agrega fila). Ver módulo
 *     "ANTI-DUPLICADOS" más abajo.
 *  7. Guarda los archivos en Drive: <Empresa>/<Año - Empresa>/<MM-Mes-Año - Empresa>/<Compras|Ventas - Empresa>
 *     y agrega una fila por archivo en "<Empresa> - Registros DTE". Lo no identificado va a "No procesado".
 *  8. Etiqueta el hilo en Gmail (<Empresa>-Compras / -Ventas / No procesado / Otros) para no reprocesarlo.
 *
 * SALIDAS:
 *  - Archivos en Drive en carpeta raíz
 *  - Una hoja "<Empresa> - Registros DTE" por empresa (columnas en SHEET_HEADERS).
 *  - Hoja "Log de Errores DTE": errores de ZIP, JSON corruptos, fallos de descarga, faltantes de
 *    pareja y duplicados omitidos.
 *  - Correo de alerta al administrador solo ante un fallo crítico general.
 *
 * CONFIGURACIÓN (sección CONFIGURACIÓN GLOBAL): empresas y NITs en COMPANIES, carpeta raíz,
 *  correo del administrador y la opción ORPHAN_REQUIRE_DTE_NAME.
 *
 * PROTECCIONES: bloqueo (LockService) para que no corran dos ejecuciones a la vez, corte seguro
 *  por tiempo, reintentos al escribir en la hoja y limpieza de JSONs con caracteres inválidos.
 *
 * HISTORIAL DE CAMBIOS (buscar "[CAMBIO v3" en el código):
 *  - v3: el PDF/XML encontrado hereda la "Fecha de Emisión DTE" de su JSON.
 *  - v3 :se quita la ruta interna del ZIP en el nombre de archivo (cleanFileName).
 *  - v3 : validación de pareja PDF + JSON con registro de faltantes.
 *  - v3 : control de duplicados por Código de Generación, nueva columna "Código de Generación"
 *    y bloqueo de ejecuciones simultáneas.
 */

function processDTEAttachments() {
  var startTime = new Date().getTime();
  
  function shouldStopExecution() {
    return (new Date().getTime() - startTime) > (CONFIG.MAX_EXECUTION_TIME_MS - CONFIG.SAFETY_MARGIN_MS);
  }

  var sheetHeaders = CONFIG.SHEET_HEADERS;
  var lock = LockService.getScriptLock();
  
  if (!lock.tryLock(10000)) {
    Logger.log('>>> [BLOQUEO] Ya hay otra ejecución en curso. Se cancela esta para evitar duplicados.');
    return;
  }

  try {
    Logger.log('>>> [INICIO] Ejecutando motor optimizado DTE');
    
    var dteMainFolder;
    try {
      dteMainFolder = DriveApp.getFolderById(CONFIG.DTE_ROOT_FOLDER_ID);
    } catch (e) {
      Logger.log('ERROR CRÍTICO: Carpeta raíz inaccesible - ID: ' + CONFIG.DTE_ROOT_FOLDER_ID);
      throw e;
    }

    var allLabels = setupLabels();
    var configs = setupCompanyConfigs(dteMainFolder, CONFIG.SHEET_HEADERS);
    var errorLogSheet = setupErrorLog(dteMainFolder);

    var threads = GmailApp.search("in:inbox has:nouserlabels");
    Logger.log('>>> [BANDEJA] ' + threads.length + ' hilos de correo por procesar.');

    var stats = { processed: 0, files: 0, duplicates: 0, others: 0, executionStopped: false };

    for (var i = 0; i < threads.length; i++) {
      if (shouldStopExecution()) {
        stats.executionStopped = true;
        Logger.log('>>> [CRONÓMETRO] Límite de tiempo alcanzado. Deteniendo ejecución de forma segura...');
        break;
      }

      var thread = threads[i];
      var threadMessages = thread.getMessages();
      var firstMsgSubject = threadMessages[0].getSubject().toLowerCase();
      var firstMsgFrom = threadMessages[0].getFrom().toLowerCase();

      // FILTRO DE SEGURIDAD: Ignorar correos del sistema, alertas de seguridad o notificaciones del propio script
      if (firstMsgSubject.includes('dte script crítico') || firstMsgSubject.includes('security alert') || firstMsgFrom.includes('no-reply@accounts.google.com')) {
        Logger.log('>>> [SKIP] Correo de sistema o alerta ignorado: "' + threadMessages[0].getSubject() + '"');
        thread.addLabel(allLabels[CONFIG.LABEL_OTROS]);
        stats.others++;
        continue;
      }

      var threadPermalink = thread.getPermalink();
      var classification = null;
      var records = [];

      for (var msgIndex = 0; msgIndex < threadMessages.length; msgIndex++) {
        var currentMessage = threadMessages[msgIndex];
        var threadData = {
          subject: currentMessage.getSubject(),
          from: currentMessage.getFrom(),
          date: currentMessage.getDate(),
          permalink: threadPermalink,
          id: currentMessage.getId()
        };
        
        Logger.log('ANALIZANDO MSG [' + msgIndex + ']: "' + threadData.subject + '" de ' + threadData.from);

        var rawAttachments = currentMessage.getAttachments();
        var processedAttachments = [];
        
        for (var a = 0; a < rawAttachments.length; a++) {
          var att = rawAttachments[a];
          var name = att.getName().toLowerCase();
          
          if (name.endsWith('.zip')) {
            Logger.log('📦 Archivo ZIP detectado: ' + att.getName() + ' -> Extrayendo contenido en memoria...');
            try {
              var extractedFiles = Utilities.unzip(att);
              for (var e = 0; e < extractedFiles.length; e++) {
                var extractedName = extractedFiles[e].getName().toLowerCase();
                if (extractedName.endsWith('.json') || extractedName.endsWith('.pdf') || extractedName.endsWith('.xml')) {
                  extractedFiles[e].setName(cleanFileName(extractedFiles[e].getName()));
                  processedAttachments.push(extractedFiles[e]);
                }
              }
            } catch (zipErr) {
              Logger.log('❌ Error descomprimiendo ZIP: ' + zipErr.toString());
              logToErrorSheet(errorLogSheet, "Error ZIP", att.getName(), threadData, zipErr.toString());
              processedAttachments.push(att); 
            }
          } else {
            processedAttachments.push(att);
          }
        }

        var nativeJsons = processedAttachments.filter(a => a.getName().toLowerCase().endsWith('.json'));
        var nativeAssets = processedAttachments.filter(a => {
          var n = a.getName().toLowerCase();
          return n.endsWith('.pdf') || n.endsWith('.xml');
        });

        var usedAssets = [];

        if (nativeJsons.length > 0) {
          Logger.log('   [A] ' + nativeJsons.length + ' JSON(s) detectado(s). Procesando lote...');
          
          nativeJsons.forEach(function(currentJson) {
            var jsonContentString = currentJson.getDataAsString('UTF-8');
            var jsonData = parseJSONSafe(jsonContentString, currentJson.getName(), errorLogSheet, threadData);
            
            if (jsonData) {
              var tempClassif = identifyCompany(jsonData);
              if (!classification || classification.name === CONFIG.FOLDER_NO_PROCESADO) {
                classification = tempClassif; 
              }
              
              records.push(createRecord(currentJson, threadData, 'json', tempClassif.label, jsonData));
              
              var rawJsonName = cleanFileName(currentJson.getName()).toLowerCase();
              var baseName = rawJsonName.substring(0, rawJsonName.lastIndexOf('.'));
              
              var matchingAsset = nativeAssets.find(a => {
                return cleanFileName(a.getName()).toLowerCase().includes(baseName);
              });

              if (!matchingAsset && nativeAssets.length === 1 && nativeJsons.length === 1) {
                matchingAsset = nativeAssets[0];
              }

              if (matchingAsset) {
                usedAssets.push(matchingAsset);
                records.push(createRecord(matchingAsset, threadData, getExt(matchingAsset.getName()), tempClassif.label, jsonData));
              } else {
                var downloaded = attemptDownloadExternalFile(currentMessage, "DTE_Complementario", threadData, errorLogSheet);
                if (downloaded && !downloaded.logError && getExt(downloaded.getName()) !== 'html') {
                  records.push(createRecord(downloaded, threadData, getExt(downloaded.getName()), tempClassif.label, jsonData, downloaded.externalUrlSource));
                } else {
                  var triedUrl = downloaded ? (downloaded.externalUrlSource || downloaded.url || '') : '';
                  records.push(buildMissingCounterpartRecord(threadData, tempClassif.label, 'PDF', currentJson.getName(), jsonData, errorLogSheet, triedUrl));
                }
              }
            }
          });

          var orphanA = registerOrphanAssets(nativeAssets, usedAssets, threadData, classification, errorLogSheet);
          records = records.concat(orphanA.records);
          classification = orphanA.classification;
        } else {
          Logger.log('   [B] No hay JSON adjunto directo. Analizando enlaces externos...');
          var scan = scanExternalLinks(currentMessage, threadData, errorLogSheet);
          if (scan.success) {
            records = records.concat(scan.records);
            if (!classification || classification.name === CONFIG.FOLDER_NO_PROCESADO) {
              classification = scan.classification;
            }
          } else if (nativeAssets.length > 0) {
            var orphanB = registerOrphanAssets(nativeAssets, [], threadData, classification, errorLogSheet);
            records = records.concat(orphanB.records);
            classification = orphanB.classification;
          }
        }
      }

      if (classification && records.length > 0) {
        var baseThreadData = {
            subject: threadMessages[0].getSubject(),
            from: threadMessages[0].getFrom(),
            date: threadMessages[0].getDate(),
            permalink: threadPermalink
        };
        var saveResult = saveThreadData(configs, classification, records, baseThreadData, CONFIG.MONTH_NAMES, errorLogSheet);
        thread.addLabel(allLabels[classification.label]);
        stats.processed++;
        stats.files += saveResult.saved;
        stats.duplicates += saveResult.duplicates;
        Logger.log('>>> [OK] Hilo procesado. Guardados: ' + saveResult.saved + ' | Duplicados: ' + saveResult.duplicates);
      } else {
        thread.addLabel(allLabels[CONFIG.LABEL_OTROS]);
        stats.others++;
        Logger.log('>>> [SKIP] Hilo sin documentos DTE válidos identificados.');
      }
    }

    SpreadsheetApp.flush();
    Logger.log('>>> [FIN] Procesamiento finalizado. Procesados: ' + stats.processed + ' | Archivos guardados: ' + stats.files);

  } catch (e) {
    Logger.log('>>> [ERROR CRÍTICO GENERAL] ' + e.toString());
    sendAdminNotification("Fallo Crítico en Script DTE", e.toString());
  } finally {
    lock.releaseLock();
  }
}

function buildMissingCounterpartRecord(t, type, missingFormat, receivedFileName, json, errorLogSheet, extUrl) {
  var fmt = missingFormat.toUpperCase();
  var received = cleanFileName(receivedFileName);
  if (errorLogSheet) {
    logToErrorSheet(errorLogSheet, 'Falta contraparte ' + fmt, received, t, 'Se recibió "' + received + '" pero no su ' + fmt);
  }
  return {
    attachment: null,
    from: t.from,
    subject: t.subject,
    fileName: fmt + ' - No encontrado (contraparte de: ' + received + ')',
    emailDate: t.date,
    fileFormat: fmt.toLowerCase(),
    type: type,
    threadPermalink: t.permalink,
    externalUrl: extUrl || 'N/A',
    fechaEmision: (json && json.identificacion) ? json.identificacion.fecEmi : '',
    codigo: getCodigoGeneracion(json) || extractGuidFromName(receivedFileName),
    isMissing: true
  };
}

function registerOrphanAssets(assets, usedAssets, threadData, classification, errorLogSheet) {
  var orphanRecords = [];
  var finalClassif = classification;

  assets.forEach(function(asset) {
    if (usedAssets.indexOf(asset) !== -1) return;
    var cleanName = cleanFileName(asset.getName());
    if (ORPHAN_REQUIRE_DTE_NAME && !looksLikeDteName(cleanName)) return;

    if (!finalClassif) {
      finalClassif = { name: CONFIG.FOLDER_NO_PROCESADO, type: 'DTE', label: CONFIG.FOLDER_NO_PROCESADO };
    }

    orphanRecords.push(createRecord(asset, threadData, getExt(asset.getName()), finalClassif.label));
    orphanRecords.push(buildMissingCounterpartRecord(threadData, finalClassif.label, 'JSON', asset.getName(), null, errorLogSheet));
  });

  return { records: orphanRecords, classification: finalClassif };
}

function identifyCompany(jsonData) {
  var emisor = getNit(jsonData, 'emisor') || getNit(jsonData, 'donante');
  var receptor = getNit(jsonData, 'receptor') || getNit(jsonData, 'donatario') || getNit(jsonData, 'documento');
  
  for (var i = 0; i < COMPANIES.length; i++) {
    var c = COMPANIES[i];
    var nitList = Array.isArray(c.nit) ? c.nit : [c.nit];
    
    if (nitList.indexOf(emisor) !== -1) return { name: c.name, type: 'Ventas', label: c.name + CONFIG.LABEL_VENTAS_SUFFIX };
    if (nitList.indexOf(receptor) !== -1) return { name: c.name, type: 'Compras', label: c.name + CONFIG.LABEL_COMPRAS_SUFFIX };
  }
  return { name: CONFIG.FOLDER_NO_PROCESADO, type: 'DTE', label: CONFIG.FOLDER_NO_PROCESADO };
}

function scanExternalLinks(message, threadData, errorLogSheet) {
  var body = message.getBody();
  var plainBody = message.getPlainBody();
  var linkRegex = /href\s*=\s*["'](https?:\/\/[^"']+)["']|(https?:\/\/[^\s"'<>]+)/gi;
  var match;
  var links = [];
  
  while ((match = linkRegex.exec(body)) !== null) {
    var url = match[1] || match[2];
    url = url.replace(/[>\]\)\.,]+$/, '');
    if (url.length > 15 && !links.includes(url) && 
        !url.includes('google.com') && !url.includes('facebook.com') && 
        !url.includes('twitter.com') && !url.includes('instagram.com') && !url.includes('schema.org') &&
        !url.toLowerCase().endsWith('.png') && !url.toLowerCase().endsWith('.jpg') && !url.toLowerCase().endsWith('.gif')) {
      links.push(url);
    }
  }

  var foundJson = null;
  var foundAsset = null;
  var jsonData = null;
  var jsonUrlSource = '';
  var assetUrlSource = '';

  for (var url of links) {
    try {
      var res = UrlFetchApp.fetch(url, { muteHttpExceptions: true, timeoutSeconds: 25, followRedirects: true });
      if (res.getResponseCode() === 200) {
        var ct = (res.getHeaders()['Content-Type'] || '').toLowerCase();
        var contentText = res.getContentText();
        var blob = res.getBlob();

        if (ct.includes('zip') || url.toLowerCase().includes('zip') || blob.getName().toLowerCase().endsWith('.zip')) {
          try {
            var extractedFiles = Utilities.unzip(blob);
            for (var e = 0; e < extractedFiles.length; e++) {
              var extName = extractedFiles[e].getName().toLowerCase();
              if (extName.endsWith('.json')) {
                var cleanContent = extractedFiles[e].getDataAsString('UTF-8').replace(/^\uFEFF/, '');
                var parsed = parseJSONSafe(cleanContent, extName, errorLogSheet, threadData);
                if (parsed) {
                  jsonData = parsed;
                  foundJson = extractedFiles[e];
                  jsonUrlSource = url;
                }
              } else if (extName.endsWith('.pdf') || extName.endsWith('.xml')) {
                foundAsset = extractedFiles[e];
                assetUrlSource = url;
              }
            }
          } catch (zipErr) {}
        } else if (!jsonData && (ct.includes('json') || contentText.trim().startsWith('{'))) {
          var cleanContent = contentText.replace(/^\uFEFF/, '');
          var parsed = parseJSONSafe(cleanContent, "url_externa_json", errorLogSheet, threadData);
          if (parsed) {
            jsonData = parsed;
            foundJson = blob;
            jsonUrlSource = url;
          }
        } else if (!foundAsset && ct.includes('pdf')) {
          foundAsset = blob;
          assetUrlSource = url;
        }
      }
    } catch (e) {}
    if (jsonData && foundAsset) break;
  }

  if (jsonData) {
    var classif = identifyCompany(jsonData);
    var cod = (jsonData.identificacion && jsonData.identificacion.codigoGeneracion) ? jsonData.identificacion.codigoGeneracion : 'DTE_Externo';
    var records = [];
    
    if (foundJson) {
      foundJson.setName(cod + ".json");
      records.push(createRecord(foundJson, threadData, 'json', classif.label, jsonData, jsonUrlSource));
    }
    
    if (foundAsset) {
      var ext = foundAsset.getName().split('.').pop().toLowerCase();
      if (ext !== 'json') {
        foundAsset.setName(cod + "." + ext);
        records.push(createRecord(foundAsset, threadData, ext, classif.label, jsonData, assetUrlSource));
      }
    } else {
      records.push(buildMissingCounterpartRecord(threadData, classif.label, 'PDF', cod + ".json", jsonData, errorLogSheet, jsonUrlSource));
    }
    return { success: true, records: records, classification: classif };
  }

  return { success: false };
}

function saveThreadData(configs, classif, records, threadData, monthNames, errorLogSheet) {
  var result = { saved: 0, duplicates: 0 };
  if (!records || records.length === 0) return result;

  var scriptTimeZone = Session.getScriptTimeZone();
  var localDate = new Date(Utilities.formatDate(threadData.date, scriptTimeZone, "yyyy-MM-dd HH:mm:ss"));
  var year = localDate.getFullYear();
  var monthName = monthNames[localDate.getMonth()];
  var monthNum = (localDate.getMonth() + 1).toString().padStart(2, '0');

  var folderCache = {};

  records.forEach(function(r) {
    var companyName = r.companyName || classif.name;
    var config = configs.find(c => c.name === companyName) || configs[0];

    if (!config) return;

    var dedupKey = buildDedupKey(r);
    var dedupIndex = dedupKey ? getDedupeIndex(config) : null;
    var missingKey = dedupKey + '|FALTANTE';
    if (dedupKey) {
      var isDuplicate = r.isMissing ? (dedupIndex[dedupKey] || dedupIndex[missingKey]) : dedupIndex[dedupKey];
      if (isDuplicate) {
        result.duplicates++;
        return;
      }
    }

    var fullTypeLabel = r.type || classif.label || '';
    var folderType = 'DTE';
    if (fullTypeLabel.indexOf(CONFIG.LABEL_VENTAS_SUFFIX) !== -1) {
      folderType = 'Ventas - ' + config.name;
    } else if (fullTypeLabel.indexOf(CONFIG.LABEL_COMPRAS_SUFFIX) !== -1) {
      folderType = 'Compras - ' + config.name;
    } else {
      folderType = CONFIG.FOLDER_NO_PROCESADO;
    }

    var cacheKey = config.name + '_' + year + '_' + monthNum + '_' + folderType;
    var targetFolder;

    if (folderCache[cacheKey]) {
      targetFolder = folderCache[cacheKey];
    } else {
      var yearDir = year + ' - ' + config.name;
      var monthDir = monthNum + '-' + monthName + '-' + year + ' - ' + config.name;

      var yearFolder = getOrCreateFolder(config.folder, yearDir);
      var monthFolder = getOrCreateFolder(yearFolder, monthDir);

      targetFolder = (folderType === CONFIG.FOLDER_NO_PROCESADO) 
        ? monthFolder 
        : getOrCreateFolder(monthFolder, folderType);

      folderCache[cacheKey] = targetFolder;
    }

    var driveUrl = "NO GUARDADO";
    if (r.attachment) {
      try {
        var file = targetFolder.createFile(r.attachment);
        driveUrl = '=HYPERLINK("' + file.getUrl() + '", "Ver Archivo")';
      } catch (e) {
        driveUrl = "ERROR DRIVE";
      }
    } else if (r.externalUrl) {
      driveUrl = r.externalUrl;
    }

    var row = [
      fullTypeLabel, 
      r.from || threadData.from, 
      r.subject || threadData.subject, 
      r.emailDate || localDate, 
      monthName, 
      year,
      r.fileName, 
      r.fileFormat, 
      driveUrl, 
      targetFolder.getName(), 
      r.fechaEmision || '',
      '=HYPERLINK("' + threadData.permalink + '", "Ver correo")', 
      new Date(), 
      r.externalUrl || '',
      r.codigo || ''
    ];

    var written = writeDataToSheetImmediately(config.sheet, [row], config.sheetName);

    if (written) {
      result.saved++;
      if (dedupKey && driveUrl !== "ERROR DRIVE") {
        dedupIndex[r.isMissing ? missingKey : dedupKey] = true;
      }
    }
  });

  return result;
}

function getCodigoGeneracion(json) {
  if (!json) return '';
  var c = (json.identificacion && json.identificacion.codigoGeneracion)
    ? json.identificacion.codigoGeneracion
    : (json.codigoGeneracion || '');
  c = c.toString().trim().toUpperCase();
  if (!c || GENERIC_CODES.indexOf(c.toLowerCase()) !== -1) return '';
  return c;
}

function extractGuidFromName(name) {
  var m = (name || '').toString().match(GUID_REGEX);
  return m ? m[0].toUpperCase() : '';
}

function buildDedupKey(r) {
  if (!r || !r.codigo) return '';
  return r.codigo + '|' + (r.fileFormat || '').toString().toLowerCase();
}

function getDedupeIndex(config) {
  if (config.dedupeIndex) return config.dedupeIndex;

  var index = {};
  var sheet = config.sheet;
  var lastRow = sheet.getLastRow();

  if (lastRow > 1) {
    var numCols = Math.min(CONFIG.SHEET_HEADERS.length, sheet.getMaxColumns());
    var values = sheet.getRange(2, 1, lastRow - 1, numCols).getValues();

    values.forEach(function(row) {
      var fileName = (row[6] || '').toString();          
      var fmt = (row[7] || '').toString().toLowerCase();      
      var driveUrl = (row[8] || '').toString();                
      var codigoCol = (row[14] || '').toString().trim().toUpperCase(); 
      var isMissing = /no encontrado/i.test(fileName);

      if (driveUrl === 'ERROR DRIVE') return; 

      var codigo = codigoCol || (isMissing ? '' : extractGuidFromName(fileName));
      if (!codigo || !fmt) return;

      index[codigo + '|' + fmt + (isMissing ? '|FALTANTE' : '')] = true;
    });
  }

  config.dedupeIndex = index;
  return index;
}

function getNit(obj, path) {
  if (!obj) return null;
  if (Array.isArray(path)) {
    for (var i = 0; i < path.length; i++) {
      var res = getNit(obj, path[i]);
      if (res) return res;
    }
    return null;
  }
  var keys = path.split('.');
  var target = obj;
  for (var k = 0; k < keys.length; k++) {
    if (target && Object.prototype.hasOwnProperty.call(target, keys[k])) {
      target = target[keys[k]];
    } else {
      target = null;
      break;
    }
  }
  if (!target) return null;
  var rawVal = (typeof target === 'object') ? (target.nit || target.numDocumento || target.numeroDocumento || '') : target;
  var val = (rawVal || '').toString().trim();
  return val ? val.replace(/-/g, '') : null;
}

function parseJSONSafe(content, name, errorLogSheet, threadData) {
  if (!content) return null;
  content = content.replace(/^\uFEFF/, '').trim();
  var cleanContent = content.replace(/\t/g, " ").replace(/[\x00-\x08\x0B-\x0C\x0E-\x1F]/g, "");
  try {
    return JSON.parse(cleanContent);
  } catch (e) {
    var lastBrace = cleanContent.lastIndexOf('}');
    while (lastBrace !== -1) {
      try {
        return JSON.parse(cleanContent.substring(0, lastBrace + 1));
      } catch (err) {
        lastBrace = cleanContent.lastIndexOf('}', lastBrace - 1);
      }
    }
  }
  return null;
}

function writeDataToSheetImmediately(sheet, rows, name) {
  var maxRetries = 3;
  var retryDelay = 1000;
  for (var attempt = 1; attempt <= maxRetries; attempt++) {
    try {
      var last = sheet.getLastRow();
      if (last + rows.length > sheet.getMaxRows()) sheet.insertRowsAfter(sheet.getMaxRows(), rows.length);
      sheet.getRange(last + 1, 1, rows.length, rows[0].length).setValues(rows);
      SpreadsheetApp.flush();
      return true;
    } catch (e) {
      if (attempt < maxRetries) {
        Utilities.sleep(retryDelay);
        retryDelay *= 2;
      }
    }
  }
  return false;
}

function attemptDownloadExternalFile(message, baseName, threadData, errorLogSheet) {
  var body = message.getPlainBody();
  var match = body.match(/(https?:\/\/[^\s\n]{5,})/i);
  if (!match) return null;
  var url = match[0].replace(/[>\]\)]+$/, '');
  try {
    var res = UrlFetchApp.fetch(url, { muteHttpExceptions: true, followRedirects: true, timeoutSeconds: 30 });
    var code = res.getResponseCode();
    if (code === 200) {
      var blob = res.getBlob();
      var ct = (res.getHeaders()['Content-Type'] || '').toLowerCase();
      var ext = ct.includes('pdf') ? 'pdf' : (ct.includes('zip') ? 'zip' : 'html');
      blob.setName(baseName + "." + ext);
      blob.externalUrlSource = url;
      return blob;
    } else {
      logToErrorSheet(errorLogSheet, "HTTP " + code, url, threadData, "Fallo descarga PDF externo.");
      return { logError: true, url: url };
    }
  } catch (e) {
    logToErrorSheet(errorLogSheet, "Fetch Error", url, threadData, e.toString());
    return { logError: true, url: url };
  }
}

function quickHealthCheck() {
  Logger.log('=== VERIFICACIÓN RÁPIDA DE SALUD ===');
  var companies = ['ATS Group', 'No procesado'];
  var allHealthy = true;
  
  companies.forEach(function(company) {

    var result = { name: company, healthy: false, message: '', lastRow: 0 };
    
    try {
      var files = DriveApp.getFilesByName(sheetName);
      if (files.hasNext()) {
        var file = files.next();
        var spreadsheet = SpreadsheetApp.openById(file.getId());
        var sheet = spreadsheet.getSheets()[0];
        var lastRow = sheet.getLastRow();
        result.lastRow = lastRow;
        
        sheet.getRange(lastRow + 1, 1).setValue('TEST_HEALTH');
        sheet.getRange(lastRow + 1, 1).clearContent();
        result.healthy = true;
        result.message = 'Saludable - ' + lastRow + ' filas';
      } else {
        result.message = 'Hoja no encontrada (se creará al procesar)';
      }
    } catch (healthError) {
      allHealthy = false;
      result.message = 'Error: ' + healthError.toString();
    }
    Logger.log(company + ': ' + result.message);
  });
  return allHealthy;
}
