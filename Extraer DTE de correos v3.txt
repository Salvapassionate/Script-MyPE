var EXECUTION_DEADLINE = 0;
var CURRENT_THREAD_TAG = '';

function TimeUpError() {
  this.timeUp = true;
  this.message = 'Tiempo de ejecución agotado';
}

function timeLeftMs() {
  return EXECUTION_DEADLINE - new Date().getTime();
}

/** Corta el hilo en curso si no queda tiempo para la siguiente operación. */
function ensureTime(neededMs) {
  if (timeLeftMs() < (neededMs || 0)) throw new TimeUpError();
}

function secondsSince(startMs) {
  return ((new Date().getTime() - startMs) / 1000).toFixed(1) + ' s';
}

function processDTEAttachments() {
  var runStart = new Date().getTime();
  EXECUTION_DEADLINE = new Date().getTime() + CONFIG.MAX_EXECUTION_TIME_MS - CONFIG.SAFETY_MARGIN_MS;

  var lock = LockService.getScriptLock();
  if (!lock.tryLock(10000)) {
    Logger.log('>>> [BLOQUEO] Ya hay otra ejecución en curso.');
    return;
  }

  try {
    Logger.log('>>> [INICIO] Ejecutando motor DTE');

    var stepStart = new Date().getTime();
    var allLabels = setupLabels();
    Logger.log('>>> [PREPARACIÓN] Etiquetas de Gmail listas (' + Object.keys(allLabels).length + ') - ' + secondsSince(stepStart));

    stepStart = new Date().getTime();
    var dteMainFolder = getOrCreateRootFolder();
    var configs = setupCompanyConfigs(dteMainFolder, CONFIG.SHEET_HEADERS);
    var errorLogSheet = setupErrorLog(dteMainFolder);
    Logger.log('>>> [PREPARACIÓN] Carpetas y hojas listas (' + configs.length + ' hojas de registro) - ' + secondsSince(stepStart));
    var ctx = { labels: allLabels, configs: configs, errorLogSheet: errorLogSheet, threadCodeMeta: {} };

    var searchQuery = CONFIG.GMAIL_SEARCH_QUERY || buildGmailSearchQuery(allLabels);
    Logger.log('>>> [BANDEJA] Query: ' + searchQuery);

    var stats = { threads: 0, processed: 0, files: 0, duplicates: 0, others: 0, failed: 0, executionStopped: false };
    var pageSize = CONFIG.GMAIL_PAGE_SIZE || 100;
    // Los hilos etiquetados salen de la búsqueda; offset solo avanza por los que quedan sin etiqueta.
    var offset = 0;
    var page = 0;

    pageLoop:
    while (true) {
      var threads = GmailApp.search(searchQuery, offset, pageSize);
      if (threads.length === 0) {
        if (page === 0) Logger.log('>>> [BANDEJA] No hay correos nuevos por revisar.');
        break;
      }
      page++;
      Logger.log('>>> [BANDEJA] Página ' + page + ': ' + threads.length + ' hilos por revisar' +
        (threads.length === pageSize ? ' (puede haber más en la siguiente página)' : '') +
        ' | Tiempo disponible: ' + Math.max(0, Math.round(timeLeftMs() / 1000)) + ' s');

      for (var i = 0; i < threads.length; i++) {
        if (timeLeftMs() <= 0) {
          stats.executionStopped = true;
          break pageLoop;
        }

        var thread = threads[i];
        var position = '[' + (i + 1) + '/' + threads.length + ']';
        CURRENT_THREAD_TAG = position;
        stats.threads++;

        if (threadHasOwnLabel(thread, allLabels)) {
          Logger.log(position + ' Ya clasificado, se omite: "' + thread.getFirstMessageSubject() + '"');
          offset++;
          continue;
        }

        var labeled = false;
        var threadStart = new Date().getTime();
        var filesBefore = stats.files;
        var duplicatesBefore = stats.duplicates;
        try {
          labeled = processThread(thread, ctx, stats);
          clearThreadTimeouts(thread);
          Logger.log(position + ' "' + thread.getFirstMessageSubject() + '" -> ' + getLabelPath(ctx.lastLabel) +
            (stats.files > filesBefore ? ' | Guardados: ' + (stats.files - filesBefore) : '') +
            (stats.duplicates > duplicatesBefore ? ' | Ya registrados antes: ' + (stats.duplicates - duplicatesBefore) : '') +
            (labeled ? '' : ' | ⚠️ NO SE PUDO ETIQUETAR') + ' | ' + secondsSince(threadStart));
        } catch (threadErr) {
          if (threadErr.timeUp) {
            // Lo guardado hasta aquí no se duplica: el anti-duplicados lo omite al reintentar.
            registerThreadTimeout(thread, new Date().getTime() - threadStart);
            stats.executionStopped = true;
            break pageLoop;
          }
          stats.failed++;
          Logger.log(position + ' [ERROR] "' + thread.getFirstMessageSubject() + '": ' + threadErr.toString());
          logToErrorSheet(errorLogSheet, 'Error procesando hilo', thread.getFirstMessageSubject(),
            { from: '', subject: thread.getFirstMessageSubject(), permalink: thread.getPermalink() }, threadErr.toString());
        }
        if (!labeled) offset++;
      }

      Logger.log('>>> [PROGRESO] Revisados: ' + stats.threads + ' | Con DTE: ' + stats.processed +
        ' | Archivos guardados: ' + stats.files + ' | Tiempo disponible: ' + Math.max(0, Math.round(timeLeftMs() / 1000)) + ' s');
    }

    SpreadsheetApp.flush();
    Logger.log('>>> [FIN] Hilos revisados: ' + stats.threads + ' | Con DTE: ' + stats.processed +
      ' | Otros: ' + stats.others + ' | Archivos guardados: ' + stats.files +
      ' | Ya registrados antes (omitidos): ' + stats.duplicates + ' | Con error: ' + stats.failed +
      ' | Duración: ' + secondsSince(runStart) +
      (stats.executionStopped ? ' | Quedan hilos pendientes para la próxima ejecución.' : ' | Buzón al día.'));

  } catch (e) {
    Logger.log('>>> [ERROR CRÍTICO GENERAL] ' + e.toString());
    sendAdminNotification("Fallo Crítico en Script DTE", e.toString());
  } finally {
    lock.releaseLock();
  }
}

/** Procesa un hilo completo. Devuelve true si quedó etiquetado. */
function processThread(thread, ctx, stats) {
  var threadMessages = thread.getMessages();
  var firstMsgSubject = threadMessages[0].getSubject().toLowerCase();
  var firstMsgFrom = threadMessages[0].getFrom().toLowerCase();

  // Correos informativos, de sistema o reportes que no son DTE.
  var ignoredSubject = CONFIG.IGNORED_SUBJECTS.some(function(t) { return firstMsgSubject.includes(t); });
  if (ignoredSubject || firstMsgFrom.includes('no-reply@accounts.google.com')) {
    stats.others++;
    ctx.lastLabel = CONFIG.LABEL_OTROS;
    return applyThreadLabel(thread, ctx.labels, ctx.lastLabel);
  }

  if (getThreadTimeouts(thread) >= CONFIG.MAX_THREAD_TIMEOUTS) {
    logToErrorSheet(ctx.errorLogSheet, 'Hilo demasiado largo', thread.getFirstMessageSubject(),
      { from: threadMessages[0].getFrom(), subject: thread.getFirstMessageSubject(), permalink: thread.getPermalink() },
      'Agotó el tiempo de ejecución ' + CONFIG.MAX_THREAD_TIMEOUTS + ' veces. Revisar manualmente.');
    stats.failed++;
    ctx.lastLabel = CONFIG.FOLDER_NO_PROCESADO;
    return applyThreadLabel(thread, ctx.labels, ctx.lastLabel);
  }

  var threadPermalink = thread.getPermalink();
  var records = [];
  ctx.threadCodeMeta = {};

  threadMessages.forEach(function(message) {
    ensureTime(0);
    var threadData = {
      subject: message.getSubject(),
      from: message.getFrom(),
      date: message.getDate(),
      permalink: threadPermalink
    };
    var assets = expandAttachments(message.getAttachments(), threadData, ctx.errorLogSheet);
    records = records.concat(analyzeMessage(message, assets, threadData, ctx));
  });

  if (records.length > 0) {
    var baseThreadData = {
      subject: threadMessages[0].getSubject(),
      from: threadMessages[0].getFrom(),
      date: threadMessages[0].getDate(),
      permalink: threadPermalink
    };
    var saveResult = saveThreadData(ctx, records, baseThreadData);
    stats.processed++;
    stats.files += saveResult.saved;
    stats.duplicates += saveResult.duplicates;
  } else {
    stats.others++;
  }

  ctx.lastLabel = resolveThreadLabel(records);
  return applyThreadLabel(thread, ctx.labels, ctx.lastLabel);
}

/** Veces que un hilo agotó el tiempo (en Propiedades del script). */
function getThreadTimeouts(thread) {
  return Number(PropertiesService.getScriptProperties().getProperty('timeout_' + thread.getId()) || 0);
}

/** Solo cuenta si el propio hilo consumió tiempo (no si empezó cuando ya quedaba poco). */
function registerThreadTimeout(thread, elapsedMs) {
  Logger.log('>>> [TIEMPO] Se agotó el tiempo en: "' + thread.getFirstMessageSubject() + '" (' + Math.round(elapsedMs / 1000) + ' s)');
  if (elapsedMs < 60 * 1000) return;
  PropertiesService.getScriptProperties().setProperty('timeout_' + thread.getId(), String(getThreadTimeouts(thread) + 1));
}

function clearThreadTimeouts(thread) {
  var props = PropertiesService.getScriptProperties();
  if (props.getProperty('timeout_' + thread.getId())) props.deleteProperty('timeout_' + thread.getId());
}

/**
 * Lista plana de archivos DTE candidatos ({ blob, name, fmt, dir }), descomprimiendo ZIPs
 * (también anidados). dir identifica el ZIP y la carpeta de origen para emparejar archivos.
 */
function expandAttachments(rawAttachments, threadData, errorLogSheet) {
  var assets = [];

  function addBlob(blob, depth, parentDir) {
    var fullName = (blob.getName() || '').toString();
    var name = cleanFileName(fullName);
    var fmt = getExt(name);
    var dir = parentDir + fullName.substring(0, fullName.length - name.length);

    if (!name || isJunkZipEntry(fullName)) return;

    if (isZipAttachment(blob, fmt)) {
      if (depth > 2) return;
      try {
        Utilities.unzip(blob).forEach(function(inner) { addBlob(inner, depth + 1, dir + name + '/'); });
      } catch (zipErr) {
        logToErrorSheet(errorLogSheet, "Error ZIP", name, threadData, zipErr.toString());
      }
      return;
    }

    if (fmt === 'json' || fmt === 'pdf' || fmt === 'xml') {
      if (depth > 0) blob.setName(name);
      assets.push({ blob: blob, name: name, fmt: fmt, dir: dir });
    }
  }

  rawAttachments.forEach(function(att) { addBlob(att, 0, ''); });
  return assets;
}

/** ZIP por extensión, por tipo de contenido o, si no tiene extensión conocida, por su firma. */
function isZipAttachment(blob, fmt) {
  if (fmt === 'zip') return true;
  if (fmt === 'json' || fmt === 'pdf' || fmt === 'xml') return false;
  var type = '';
  try { type = (blob.getContentType() || '').toLowerCase(); } catch (e) {}
  if (type.indexOf('zip') !== -1) return true;
  return (type === '' || type.indexOf('octet-stream') !== -1) && isZipBlob(blob);
}

/** Entradas que no son documentos: carpetas y metadatos de macOS/Windows. */
function isJunkZipEntry(path) {
  var name = cleanFileName(path);
  return /(^|\/)__MACOSX\//i.test(path) || name.indexOf('._') === 0 || /^(\.ds_store|thumbs\.db)$/i.test(name);
}

/** Texto de un archivo respetando BOM UTF-8 / UTF-16. */
function readBlobText(blob) {
  var bytes = blob.getBytes();
  if (bytes.length >= 2 && (bytes[0] & 0xFF) === 0xFF && (bytes[1] & 0xFF) === 0xFE) return blob.getDataAsString('UTF-16LE').replace(/^\uFEFF/, '');
  if (bytes.length >= 2 && (bytes[0] & 0xFF) === 0xFE && (bytes[1] & 0xFF) === 0xFF) return blob.getDataAsString('UTF-16BE').replace(/^\uFEFF/, '');
  return blob.getDataAsString('UTF-8');
}

/**
 * Registros de un mensaje: cada JSON se clasifica por separado y se empareja con su PDF/XML
 * (por nombre, por Código de Generación en el nombre o en el contenido, o por descarte).
 */
function analyzeMessage(message, assets, threadData, ctx) {
  var records = [];
  var fetchCache = {};
  var jsonAssets = assets.filter(function(a) { return a.fmt === 'json'; });
  var pending = assets.filter(function(a) { return a.fmt === 'pdf' || a.fmt === 'xml'; });
  var hasHints = messageHasDteHints(message, assets);
  var docs = [];

  jsonAssets.forEach(function(a) {
    var parsed = parseJSONSafe(readBlobText(a.blob));
    var dte = unwrapDte(parsed);
    if (!dte) {
      logToErrorSheet(ctx.errorLogSheet, parsed ? 'JSON no es DTE' : 'JSON corrupto', a.name, threadData,
        parsed ? 'No se encontró identificación, emisor ni receptor.' : 'No se pudo leer el contenido.');
      return;
    }
    docs.push({ asset: a, meta: buildDteMeta(dte, a.name), pairs: {} });
  });

  // Sin JSON adjunto: buscar el DTE en los enlaces del correo.
  if (jsonAssets.length === 0 && hasHints) {
    var ext = scanExternalLinks(message, threadData, ctx.errorLogSheet, fetchCache);
    if (ext) {
      if (ext.dte) {
        var meta = buildDteMeta(ext.dte, '');
        var cod = meta.codigo || 'DTE_Externo';
        ext.jsonBlob.setName(cod + '.json');
        docs.push({ asset: { blob: ext.jsonBlob, name: cod + '.json', fmt: 'json', url: ext.url }, meta: meta, pairs: {} });
        if (ext.asset) {
          ext.asset.name = cod + '.' + ext.asset.fmt;
          ext.asset.blob.setName(ext.asset.name);
        }
      }
      if (ext.asset) {
        if (!ext.asset.name) {
          var extName = cleanFileName(ext.asset.blob.getName()) || 'DTE_Externo';
          ext.asset.name = getExt(extName) === ext.asset.fmt ? extName : extName + '.' + ext.asset.fmt;
          ext.asset.blob.setName(ext.asset.name);
        }
        pending.push(ext.asset);
      }
    }
  }

  docs.forEach(function(doc) { rememberCodeMeta(ctx, doc.meta); });

  // Emparejar primero por Código de Generación en el nombre (lo más seguro).
  docs.forEach(function(doc) {
    var code = (doc.meta.codigo || '').toLowerCase();
    if (!code) return;
    pending = pending.filter(function(a) {
      if (doc.pairs[a.fmt] || a.name.toLowerCase().indexOf(code) === -1) return true;
      doc.pairs[a.fmt] = a;
      return false;
    });
  });

  // Luego por nombre: primero dentro de la misma carpeta/ZIP de origen, después en cualquiera.
  [true, false].forEach(function(sameDirOnly) {
    docs.forEach(function(doc) {
      var base = baseName(doc.asset.name).toLowerCase();
      pending = pending.filter(function(a) {
        if (doc.pairs[a.fmt]) return true;
        if (sameDirOnly && (a.dir || '') !== (doc.asset.dir || '')) return true;
        var n = a.name.toLowerCase();
        var sameName = baseName(n) === base || (base.length >= 6 && n.indexOf(base) !== -1);
        if (!sameName) return true;
        doc.pairs[a.fmt] = a;
        return false;
      });
    });
  });

  // Emparejar por Código de Generación dentro del contenido.
  var docsWithoutPair = function(fmt) { return docs.filter(function(d) { return !d.pairs[fmt] && d.meta.codigo; }); };
  if (pending.length > 0 && (docsWithoutPair('pdf').length > 0 || docsWithoutPair('xml').length > 0)) {
    pending = pending.filter(function(a) {
      var candidates = docsWithoutPair(a.fmt);
      if (candidates.length === 0) return true;
      var codes = extractInfoFromText(getAssetText(a)).codigos;
      var doc = candidates.find(function(d) { return codes.indexOf(d.meta.codigo) !== -1; });
      if (!doc) return true;
      doc.pairs[a.fmt] = a;
      return false;
    });
  }

  // Por descarte: un solo JSON y un solo archivo de ese formato.
  ['pdf', 'xml'].forEach(function(fmt) {
    var lonelyDocs = docs.filter(function(d) { return !d.pairs[fmt]; });
    var lonelyAssets = pending.filter(function(a) { return a.fmt === fmt; });
    if (lonelyDocs.length === 1 && lonelyAssets.length === 1 && docs.length === 1) {
      lonelyDocs[0].pairs[fmt] = lonelyAssets[0];
      pending.splice(pending.indexOf(lonelyAssets[0]), 1);
    }
  });

  var docsMissingPdf = docs.filter(function(d) { return !d.pairs.pdf; });
  docs.forEach(function(doc) {
    records.push(createRecord(doc.asset, threadData, doc.meta, doc.asset.url));
    if (doc.pairs.xml) records.push(createRecord(doc.pairs.xml, threadData, doc.meta, doc.pairs.xml.url));

    if (doc.pairs.pdf) {
      records.push(createRecord(doc.pairs.pdf, threadData, doc.meta, doc.pairs.pdf.url));
      return;
    }

    if (isAlreadyRegistered(ctx, doc.meta.codigo, 'pdf')) return;

    var detail = '';
    // Solo se descarga si hay un único JSON sin PDF (si no, no se sabe a cuál pertenece).
    if (docsMissingPdf.length === 1) {
      var downloaded = attemptDownloadExternalFile(message, doc.meta.codigo || 'DTE_Complementario', fetchCache);
      if (downloaded && downloaded.blob) {
        records.push(createRecord({ blob: downloaded.blob, name: downloaded.blob.getName(), fmt: 'pdf' }, threadData, doc.meta, downloaded.url));
        return;
      }
      if (downloaded) detail = downloaded.detail;
    }
    records.push(buildMissingCounterpartRecord(threadData, doc.meta, 'PDF', doc.asset.name, detail));
  });

  // Los XML primero: su contenido clasifica el documento y el PDF del mismo código lo hereda.
  var uniqueClassif = getUniqueCompanyClassification(docs);
  pending.sort(function(x, y) { return (x.fmt === 'xml' ? 0 : 1) - (y.fmt === 'xml' ? 0 : 1); });
  pending.forEach(function(a) {
    records = records.concat(buildOrphanRecords(a, threadData, ctx, hasHints, uniqueClassif, docs.length > 0));
  });

  return records;
}

function buildDteMeta(dte, fileName) {
  var tipoCodigo = getTipoDte(dte);
  return {
    codigo: getCodigoGeneracion(dte) || extractGuidFromName(fileName),
    fecha: getFechaEmision(dte),
    tipoNombre: getTipoDteNombre(dte, tipoCodigo),
    classif: identifyCompany(dte, tipoCodigo)
  };
}

/** Devuelve el objeto DTE aunque venga envuelto en otro objeto, arreglo o texto. */
function unwrapDte(value, depth) {
  depth = depth || 0;
  if (value === null || value === undefined || depth > 4) return null;

  if (typeof value === 'string') {
    var s = value.trim();
    if (s.charAt(0) !== '{' && s.charAt(0) !== '[') return null;
    return unwrapDte(parseJSONSafe(s), depth + 1);
  }

  if (Array.isArray(value)) {
    for (var i = 0; i < value.length; i++) {
      var fromItem = unwrapDte(value[i], depth + 1);
      if (fromItem) return fromItem;
    }
    return null;
  }

  if (typeof value !== 'object') return null;
  if (isDteObject(value)) return value;

  for (var key in value) {
    if (!Object.prototype.hasOwnProperty.call(value, key)) continue;
    var child = value[key];
    if (child && (typeof child === 'object' || typeof child === 'string')) {
      var found = unwrapDte(child, depth + 1);
      if (found) return found;
    }
  }
  return null;
}

function isDteObject(obj) {
  var ident = getKeyCI(obj, 'identificacion');
  if (ident && typeof ident === 'object') return true;
  var emisor = getKeyCI(obj, 'emisor') || getKeyCI(obj, 'donatario');
  return !!(emisor && typeof emisor === 'object');
}

/** Lee una propiedad sin importar mayúsculas ni guiones bajos. */
function getKeyCI(obj, key) {
  if (!obj || typeof obj !== 'object') return undefined;
  if (Object.prototype.hasOwnProperty.call(obj, key)) return obj[key];
  var wanted = key.toLowerCase().replace(/_/g, '');
  for (var k in obj) {
    if (Object.prototype.hasOwnProperty.call(obj, k) && k.toLowerCase().replace(/_/g, '') === wanted) return obj[k];
  }
  return undefined;
}

function getPathCI(obj, path) {
  var keys = path.split('.');
  var target = obj;
  for (var i = 0; i < keys.length; i++) {
    target = getKeyCI(target, keys[i]);
    if (target === undefined || target === null) return undefined;
  }
  return target;
}

function findKeyDeep(obj, keyRegex, depth) {
  depth = depth || 0;
  if (!obj || typeof obj !== 'object' || depth > 3) return undefined;
  for (var k in obj) {
    if (!Object.prototype.hasOwnProperty.call(obj, k)) continue;
    if (keyRegex.test(k) && obj[k] !== null && obj[k] !== '' && typeof obj[k] !== 'object') return obj[k];
  }
  for (var k2 in obj) {
    if (!Object.prototype.hasOwnProperty.call(obj, k2)) continue;
    var found = findKeyDeep(obj[k2], keyRegex, depth + 1);
    if (found !== undefined) return found;
  }
  return undefined;
}

function getFechaEmision(dte) {
  if (!dte) return '';
  var paths = [
    'identificacion.fecEmi', 'identificacion.fechaEmision', 'identificacion.fecha',
    'fecEmi', 'fechaEmision', 'documento.fecEmi', 'identificacion.fecAnula'
  ];
  for (var i = 0; i < paths.length; i++) {
    var d = parseEmissionDate(getPathCI(dte, paths[i]));
    if (d) return formatYmd(d);
  }
  var deep = parseEmissionDate(findKeyDeep(dte, /^fec(ha)?_?emi(sion)?$/i));
  return deep ? formatYmd(deep) : '';
}

function getCodigoGeneracion(json) {
  if (!json) return '';
  var c = getPathCI(json, 'identificacion.codigoGeneracion') || getKeyCI(json, 'codigoGeneracion') ||
    findKeyDeep(json, /^codigo_?generacion$/i) || '';
  c = c.toString().trim().toUpperCase();
  if (!c || GENERIC_CODES.indexOf(c.toLowerCase()) !== -1) return '';
  return c;
}

function getTipoDte(dte) {
  var t = getPathCI(dte, 'identificacion.tipoDte') || getKeyCI(dte, 'tipoDte') || getPathCI(dte, 'documento.tipoDte') || '';
  t = t.toString().trim();
  return /^\d$/.test(t) ? '0' + t : t;
}

function getTipoNombre(tipoCodigo) {
  return tipoCodigo ? tipoCodigo + ' - ' + (DTE_TYPE_NAMES[tipoCodigo] || 'Tipo desconocido') : '';
}

function getTipoDteNombre(dte, tipoCodigo) {
  var name = getTipoNombre(tipoCodigo);
  var documento = getKeyCI(dte, 'documento');
  var esAnulacion = getKeyCI(dte, 'motivo') && documento && typeof documento === 'object';
  return esAnulacion ? 'Anulación' + (name ? ' de ' + name : '') : name;
}

/** NIT/DUI como dígitos sin ceros iniciales; null si no lo parece. */
function normalizeNit(value) {
  if (value === null || value === undefined || typeof value === 'object') return null;
  var digits = value.toString().replace(/\D/g, '');
  if (digits.length < 8 || digits.length > 14) return null;
  return digits.replace(/^0+/, '');
}

function extractNitValue(node, depth) {
  depth = depth || 0;
  if (node === null || node === undefined || depth > 3) return null;
  if (typeof node !== 'object') return normalizeNit(node);

  for (var i = 0; i < NIT_KEYS.length; i++) {
    var v = getKeyCI(node, NIT_KEYS[i]);
    if (v === undefined || v === null || v === '') continue;
    var nit = extractNitValue(v, depth + 1);
    if (nit) return nit;
  }
  return null;
}

function getPartyNit(jsonData, partyNames) {
  if (!jsonData) return null;
  for (var i = 0; i < partyNames.length; i++) {
    var party = getKeyCI(jsonData, partyNames[i]);
    if (!party || typeof party !== 'object') continue;
    var nit = extractNitValue(party);
    if (nit) return nit;
  }
  return null;
}

function getCompanyNits(company) {
  var list = company.nits || company.nit || [];
  return (Array.isArray(list) ? list : [list]).map(normalizeNit).filter(Boolean);
}

function makeClassification(company, type) {
  var suffix = type === 'Ventas' ? CONFIG.LABEL_VENTAS_SUFFIX : CONFIG.LABEL_COMPRAS_SUFFIX;
  return { name: company.name, type: type, label: company.name + suffix };
}

}

}

function identifyCompany(jsonData, tipoCodigo) {
  var inverted = CONFIG.INVERTED_DTE_TYPES.indexOf(tipoCodigo) !== -1;
  var emisor = getPartyNit(jsonData, ['emisor', 'donatario']);
  var receptor = getPartyNit(jsonData, ['receptor', 'cliente', 'sujetoExcluido', 'donante', 'documento']);

  for (var i = 0; i < COMPANIES.length; i++) {
    var nitList = getCompanyNits(COMPANIES[i]);
    if (emisor && nitList.indexOf(emisor) !== -1) return makeClassification(COMPANIES[i], inverted ? 'Compras' : 'Ventas');
    if (receptor && nitList.indexOf(receptor) !== -1) return makeClassification(COMPANIES[i], inverted ? 'Ventas' : 'Compras');
  }
  return noProcesadoClassification();
}

/** Texto de un XML. Los PDF no se leen (sin OCR): devuelve vacío. */
function getAssetText(asset) {
  if (asset.text !== undefined) return asset.text;
  try {
    asset.text = asset.fmt === 'pdf' ? '' : readBlobText(asset.blob);
  } catch (e) {
    asset.text = '';
  }
  return asset.text;
}

/** Códigos de Generación, fecha de emisión, tipo y NITs (en orden de aparición) de un texto. */
function extractInfoFromText(text) {
  var info = { codigos: [], fecha: '', nits: [], tipoCodigo: '' };
  if (!text) return info;

  var tipoMatch = text.match(/DTE-(\d{2})-[A-Z0-9]{8}-\d{15}/i);
  if (tipoMatch) info.tipoCodigo = tipoMatch[1];

  var m;
  var guidRe = new RegExp(GUID_REGEX.source, 'gi');
  while ((m = guidRe.exec(text)) !== null) {
    var c = m[0].toUpperCase();
    if (info.codigos.indexOf(c) === -1) info.codigos.push(c);
  }

  var datePatterns = [
    /<fecEmi>\s*([^<]+)</i,
    /"fecEmi"\s*:\s*"([^"]+)"/i,
    /(\d{4}-\d{1,2}-\d{1,2}|\d{1,2}[\/-]\d{1,2}[\/-]\d{4})[^0-9]{0,40}hora\s*(?:de\s*)?emisi[oó]n/i,
    /fecha\s*(?:y\s*hora\s*)?(?:de\s*)?emisi[oó]n[^0-9]{0,80}(\d{4}-\d{1,2}-\d{1,2}|\d{1,2}[\/-]\d{1,2}[\/-]\d{4})/i,
    /(\d{4}-\d{2}-\d{2}|\d{1,2}\/\d{1,2}\/\d{4})/
  ];
  for (var i = 0; i < datePatterns.length; i++) {
    var dm = text.match(datePatterns[i]);
    var d = dm ? parseEmissionDate(dm[1]) : null;
    if (d) {
      info.fecha = formatYmd(d);
      break;
    }
  }

  var nitRe = /\b\d{4}-\d{6}-\d{3}-\d\b|\b\d{14}\b|\b\d{8}-\d\b/g;
  while ((m = nitRe.exec(text)) !== null) {
    var nit = normalizeNit(m[0]);
    if (nit && info.nits.indexOf(nit) === -1) info.nits.push(nit);
  }
  return info;
}

/** El emisor aparece antes que el receptor: si el NIT de la empresa es el primero, es Venta. */
function identifyCompanyFromNits(nits) {
  if (!nits || nits.length === 0) return null;
  var best = null;
  COMPANIES.forEach(function(c) {
    getCompanyNits(c).forEach(function(n) {
      var idx = nits.indexOf(n);
      if (idx !== -1 && (!best || idx < best.idx)) best = { idx: idx, company: c };
    });
  });
  if (!best) return null;
  return makeClassification(best.company, best.idx === 0 ? 'Ventas' : 'Compras');
}

/** Datos de un DTE en XML (mismos campos que el JSON del Ministerio de Hacienda). */
function buildXmlDteMeta(text, fileName) {
  if (!text) return null;

  function block(tag) {
    var m = text.match(new RegExp('<(?:\\w+:)?' + tag + '\\b[^>]*>([\\s\\S]*?)<\\/(?:\\w+:)?' + tag + '>', 'i'));
    return m ? m[1] : '';
  }
  function value(source, tags) {
    for (var i = 0; i < tags.length; i++) {
      var m = source.match(new RegExp('<(?:\\w+:)?' + tags[i] + '\\b[^>]*>\\s*([^<]+?)\\s*<', 'i'));
      if (m) return m[1];
    }
    return '';
  }

  var idBlock = block('identificacion') || text;
  var nitTags = ['nit', 'numDocumento', 'numeroDocumento', 'dui', 'documento'];
  var dte = {
    identificacion: {
      codigoGeneracion: value(idBlock, ['codigoGeneracion']),
      fecEmi: value(idBlock, ['fecEmi', 'fechaEmision']),
      tipoDte: value(idBlock, ['tipoDte'])
    },
    emisor: { nit: value(block('emisor') || block('donatario'), nitTags) },
    receptor: { nit: value(block('receptor') || block('cliente') || block('sujetoExcluido') || block('donante'), nitTags) }
  };
  if (!dte.emisor.nit && !dte.identificacion.codigoGeneracion) return null;
  return buildDteMeta(dte, fileName);
}

/**
 * Registros de un PDF/XML sin JSON. El XML se clasifica por su contenido. El PDF hereda los datos
 * si su código ya se registró; si no, queda como "Solo PDF" para revisión manual. Un PDF sin
 * código en el nombre solo se registra si el correo no trae JSON y parece de DTE.
 */
function buildOrphanRecords(asset, threadData, ctx, hasHints, fallbackClassif, messageHasJson) {
  var text = asset.fmt === 'xml' ? getAssetText(asset) : '';
  var info = extractInfoFromText(text);
  var xmlMeta = asset.fmt === 'xml' ? buildXmlDteMeta(text, asset.name) : null;
  var codigo = (xmlMeta && xmlMeta.codigo) || extractGuidFromName(asset.name) || info.codigos[0] || '';
  var probablePdfDte = asset.fmt === 'pdf' && hasHints && !messageHasJson;
  if (ORPHAN_REQUIRE_DTE_NAME && !codigo && !probablePdfDte) return [];

  // Si el JSON (o XML) de este código ya se leyó, se heredan sus datos.
  var known = codigo ? lookupCodeMeta(ctx, codigo) : null;
  var xmlClassif = xmlMeta && xmlMeta.classif.type !== 'DTE' ? xmlMeta.classif : null;
  var meta = {
    codigo: codigo,
    fecha: (known && known.fecha) || (xmlMeta && xmlMeta.fecha) || info.fecha || '',
    tipoNombre: (known && known.tipoNombre) || (xmlMeta && xmlMeta.tipoNombre) || getTipoNombre(info.tipoCodigo),
    classif: (known && known.classif) || xmlClassif || identifyCompanyFromNits(info.nits) || fallbackClassif ||
      (asset.fmt === 'pdf' ? soloPdfClassification() : noProcesadoClassification())
  };
  if (xmlMeta) rememberCodeMeta(ctx, meta);

  return [
    createRecord(asset, threadData, meta, asset.url),
    buildMissingCounterpartRecord(threadData, meta, 'JSON', asset.name, '')
  ];
}

/** Clasificación común si todos los JSON del mensaje coinciden; si no, null. */
function getUniqueCompanyClassification(docs) {

  docs.forEach(function(d) { if (d.meta.classif.type !== 'DTE') labels[d.meta.classif.label] = d.meta.classif; });

}

function messageHasDteHints(message, assets) {

  var text = (message.getSubject() + ' ' + message.getPlainBody()).toLowerCase();

}

function isIgnoredLink(url) {
  if (IGNORED_LINK_PATTERN.test(url)) return true;
  var hostMatch = url.match(/^https?:\/\/([^\/?#:]+)/i);
  var host = hostMatch ? hostMatch[1].toLowerCase() : '';
  return IGNORED_LINK_DOMAINS.some(function(d) { return host === d || host.endsWith('.' + d); });
}

function extractCandidateLinks(message) {
  var sources = [message.getBody() || '', message.getPlainBody() || ''];
  var linkRegex = /href\s*=\s*["'](https?:\/\/[^"']+)["']|(https?:\/\/[^\s"'<>]+)/gi;
  var links = [];

  sources.forEach(function(src) {
    var match;
    linkRegex.lastIndex = 0;
    while ((match = linkRegex.exec(src)) !== null) {
      var url = (match[1] || match[2]).replace(/&amp;/g, '&').replace(/[>\]\)\.,;]+$/, '');
      if (url.length > 15 && links.indexOf(url) === -1 && !isIgnoredLink(url)) links.push(url);
    }
  });

  links.sort(function(a, b) { return (PREFERRED_LINK_PATTERN.test(b) ? 1 : 0) - (PREFERRED_LINK_PATTERN.test(a) ? 1 : 0); });
  return links.slice(0, CONFIG.MAX_LINKS_PER_MESSAGE || 6);
}

function fetchUrlCached(url, cache) {
  if (Object.prototype.hasOwnProperty.call(cache, url)) return cache[url];
  ensureTime(CONFIG.FETCH_RESERVE_MS);
  var hostMatch = url.match(/^https?:\/\/([^\/?#:]+)/i);
  Logger.log(CURRENT_THREAD_TAG + ' Revisando enlace externo: ' + (hostMatch ? hostMatch[1] : url));
  var out;
  try {
    var res = UrlFetchApp.fetch(url, { muteHttpExceptions: true, followRedirects: true });
    var headers = res.getHeaders();
    out = {
      code: res.getResponseCode(),
      contentType: (headers['Content-Type'] || headers['content-type'] || '').toLowerCase(),
      blob: res.getBlob(),
      error: ''
    };
  } catch (e) {
    out = { code: 0, contentType: '', blob: null, error: e.toString() };
  }
  cache[url] = out;
  return out;
}

function blobStartsWith(blob, signature) {
  try {
    var bytes = blob.getBytes();
    for (var i = 0; i < signature.length; i++) {
      if (bytes[i] !== signature[i]) return false;
    }
    return bytes.length >= signature.length;
  } catch (e) {
    return false;
  }
}

function isZipBlob(blob) { return blobStartsWith(blob, [0x50, 0x4B]); }
function isPdfBlob(blob) { return blobStartsWith(blob, [0x25, 0x50, 0x44, 0x46]); }

/** Busca en los enlaces del correo un JSON de DTE y su PDF/XML. Devuelve { dte, jsonBlob, asset, url } o null. */
function scanExternalLinks(message, threadData, errorLogSheet, fetchCache) {
  var links = extractCandidateLinks(message);
  var found = { dte: null, jsonBlob: null, asset: null, url: '' };

  for (var i = 0; i < links.length; i++) {
    var url = links[i];
    var res = fetchUrlCached(url, fetchCache);
    if (res.code !== 200 || !res.blob) continue;
    var blob = res.blob;

    if (res.contentType.indexOf('zip') !== -1 || isZipBlob(blob)) {
      var files;
      try {
        files = Utilities.unzip(blob);
      } catch (zipErr) {
        logToErrorSheet(errorLogSheet, "Error ZIP externo", url, threadData, zipErr.toString());
        continue;
      }
      files.forEach(function(f) {
        var n = f.getName().toLowerCase();
        if (!found.dte && n.endsWith('.json')) {
          var d = unwrapDte(parseJSONSafe(f.getDataAsString('UTF-8')));
          if (d) {
            found.dte = d;
            found.jsonBlob = f;
            found.url = url;
          }
        } else if (!found.asset && (n.endsWith('.pdf') || n.endsWith('.xml'))) {
          found.asset = { blob: f, fmt: getExt(n), url: url };
        }
      });
    } else if (res.contentType.indexOf('pdf') !== -1 || isPdfBlob(blob)) {
      if (!found.asset) found.asset = { blob: blob, fmt: 'pdf', url: url };
    } else if (!found.dte && (res.contentType.indexOf('json') !== -1 || res.contentType.indexOf('text') !== -1 || !res.contentType)) {
      var text = '';
      try { text = blob.getDataAsString('UTF-8').trim(); } catch (txtErr) {}
      if (text.charAt(0) === '{' || text.charAt(0) === '[') {
        var dte = unwrapDte(parseJSONSafe(text));
        if (dte) {
          found.dte = dte;
          found.jsonBlob = blob;
          found.url = url;
        }
      }
    }
    if (found.dte && found.asset) break;
  }

  return (found.dte || found.asset) ? found : null;
}

/** Busca el PDF de un JSON en los enlaces del correo. Devuelve { blob, url }, { detail } o null. */
function attemptDownloadExternalFile(message, baseFileName, fetchCache) {
  var links = extractCandidateLinks(message);
  if (links.length === 0) return null;

  var attempts = [];
  for (var i = 0; i < links.length; i++) {
    var url = links[i];
    var res = fetchUrlCached(url, fetchCache);
    if (res.code !== 200 || !res.blob) {
      attempts.push((res.error ? 'Error' : 'HTTP ' + res.code) + ': ' + url);
      continue;
    }

    if (res.contentType.indexOf('pdf') !== -1 || isPdfBlob(res.blob)) {
      res.blob.setName(baseFileName + '.pdf');
      return { blob: res.blob, url: url };
    }

    if (res.contentType.indexOf('zip') !== -1 || isZipBlob(res.blob)) {
      try {
        var pdf = Utilities.unzip(res.blob).find(function(f) { return f.getName().toLowerCase().endsWith('.pdf'); });
        if (pdf) {
          pdf.setName(baseFileName + '.pdf');
          return { blob: pdf, url: url };
        }
      } catch (zipErr) {}
    }
    attempts.push('Sin PDF: ' + url);
  }
  return { detail: 'Enlaces revisados -> ' + attempts.join(' | ') };
}

function saveThreadData(ctx, records, threadData) {
  var result = { saved: 0, duplicates: 0 };
  var scriptTimeZone = Session.getScriptTimeZone();
  var monthNames = CONFIG.MONTH_NAMES;
  var folderCache = {};

  records.forEach(function(r) {
    ensureTime(0);
    var config = findConfig(ctx.configs, r.companyName);
    if (!config) return;

    var folderDate = parseEmissionDate(r.fechaEmision);
    if (!folderDate) {
      var fallbackDate = r.emailDate ? new Date(r.emailDate) : threadData.date;
      folderDate = parseEmissionDate(Utilities.formatDate(fallbackDate, scriptTimeZone, "yyyy-MM-dd"));
    }
    var year = folderDate.getFullYear();
    var monthName = monthNames[folderDate.getMonth()];
    var monthNum = (folderDate.getMonth() + 1).toString().padStart(2, '0');

    var dedupKey = buildDedupKey(r);
    var dedupIndex = getDedupeIndex(config);
    var missingKey = dedupKey + '|FALTANTE';
    if (dedupKey) {
      var isDuplicate = r.isMissing ? (isAlreadyRegistered(ctx, r.codigo, r.fileFormat) || dedupIndex[missingKey]) : dedupIndex[dedupKey];
      if (isDuplicate) {
        result.duplicates++;
        return;
      }
    }

    var isCompany = config.name !== CONFIG.FOLDER_NO_PROCESADO;
    var folderType = CONFIG.FOLDER_NO_PROCESADO;
    if (isCompany && r.classType === 'Ventas') folderType = 'Ventas - ' + config.name;
    if (isCompany && r.classType === 'Compras') folderType = 'Compras - ' + config.name;

    var cacheKey = config.name + '_' + year + '_' + monthNum + '_' + folderType;
    var targetFolder = folderCache[cacheKey];
    if (!targetFolder) {
      var yearFolder = getOrCreateFolder(config.folder, year + ' - ' + config.name);
      var monthFolder = getOrCreateFolder(yearFolder, monthNum + '-' + monthName + '-' + year + ' - ' + config.name);
      targetFolder = (folderType === CONFIG.FOLDER_NO_PROCESADO) ? monthFolder : getOrCreateFolder(monthFolder, folderType);
      folderCache[cacheKey] = targetFolder;
    }

    var driveUrl = '';
    var createdFile = null;
    if (r.attachment) {
      try {
        createdFile = targetFolder.createFile(r.attachment);
        driveUrl = '=HYPERLINK("' + createdFile.getUrl() + '", "Ver Archivo")';
      } catch (e) {
        driveUrl = "ERROR DRIVE";
        logToErrorSheet(ctx.errorLogSheet, 'Error Drive', r.fileName, r, e.toString());
      }
    }

    var row = [
      r.typeLabel || '',
      r.from || threadData.from,
      r.subject || threadData.subject,
      r.emailDate || folderDate,
      monthName,
      year,
      r.fileName,
      r.fileFormat,
      driveUrl,
      targetFolder.getName(),
      r.fechaEmision || '',
      '=HYPERLINK("' + threadData.permalink + '", "Ver correo")',
      new Date(),
      r.externalUrl || '',
      r.codigo || '',
      r.tipoDoc || ''
    ];

    if (!writeDataToSheetImmediately(config.sheet, [row])) {
      if (createdFile) {
        try { createdFile.setTrashed(true); } catch (trashErr) {}
      }
      logToErrorSheet(ctx.errorLogSheet, 'Error escritura hoja', r.fileName, r, 'No se pudo escribir en ' + config.sheetName);
      return;
    }

    result.saved++;
    if (dedupKey && driveUrl !== "ERROR DRIVE") {
      dedupIndex[r.isMissing ? missingKey : dedupKey] = true;
    }

    if (r.isMissing) {
      logToErrorSheet(ctx.errorLogSheet, 'Falta contraparte ' + r.fileFormat.toUpperCase(), r.counterpartOf, r,
        'Se recibió "' + r.counterpartOf + '" pero no su ' + r.fileFormat.toUpperCase() +
        (r.codigo ? ' (Código: ' + r.codigo + ')' : '') + (r.missingDetail ? '. ' + r.missingDetail : ''));
    } else if (r.codigo && r.fileFormat === 'json' && !config.codeMeta[r.codigo]) {
      config.codeMeta[r.codigo] = { classif: { name: r.companyName, type: r.classType, label: r.typeLabel }, fecha: r.fechaEmision, tipoNombre: r.tipoDoc };
    }
  });

  return result;
}

function parseEmissionDate(value) {
  if (value === null || value === undefined || value === '') return null;

  if (Object.prototype.toString.call(value) === '[object Date]') {
    return isNaN(value.getTime()) ? null : new Date(value.getFullYear(), value.getMonth(), value.getDate());
  }

  if (typeof value === 'number') {
    var fromNumber = new Date(value);
    return isNaN(fromNumber.getTime()) ? null : new Date(fromNumber.getFullYear(), fromNumber.getMonth(), fromNumber.getDate());
  }

  var text = value.toString().trim();
  var iso = text.match(/^(\d{4})[-\/](\d{1,2})[-\/](\d{1,2})/);
  if (iso) return new Date(Number(iso[1]), Number(iso[2]) - 1, Number(iso[3]));

  var dmy = text.match(/^(\d{1,2})[-\/](\d{1,2})[-\/](\d{4})/);
  if (dmy) return new Date(Number(dmy[3]), Number(dmy[2]) - 1, Number(dmy[1]));

  var parsed = new Date(text);
  return isNaN(parsed.getTime()) ? null : new Date(parsed.getFullYear(), parsed.getMonth(), parsed.getDate());
}

function formatYmd(date) {
  if (!date) return '';
  return date.getFullYear() + '-' + ('0' + (date.getMonth() + 1)).slice(-2) + '-' + ('0' + date.getDate()).slice(-2);
}

function extractGuidFromName(name) {
  var m = (name || '').toString().match(GUID_REGEX);
  return m ? m[0].toUpperCase() : '';
}

function buildDedupKey(r) {
  if (!r || !r.codigo) return '';
  return r.codigo + '|' + (r.fileFormat || '').toString().toLowerCase();
}

/**
 * Lee una vez por ejecución la hoja de una empresa:
 *  dedupeIndex: "CODIGO|formato" ya registrados; codeMeta: datos del JSON de cada código.
 */
function getDedupeIndex(config) {
  if (config.dedupeIndex) return config.dedupeIndex;

  var index = {};
  var codeMeta = {};
  var sheet = config.sheet;
  var lastRow = sheet.getLastRow();

  if (lastRow > 1) {
    var numCols = Math.min(CONFIG.SHEET_HEADERS.length, sheet.getMaxColumns());
    var values = sheet.getRange(2, 1, lastRow - 1, numCols).getValues();

    values.forEach(function(row) {
      var label = (row[0] || '').toString();
      var fileName = (row[6] || '').toString();
      var fmt = (row[7] || '').toString().toLowerCase();
      var driveUrl = (row[8] || '').toString();
      var codigoCol = (row[14] || '').toString().trim().toUpperCase();
      var isMissing = /no encontrado/i.test(fileName);

      if (driveUrl === 'ERROR DRIVE') return;

      var codigo = codigoCol || (isMissing ? '' : extractGuidFromName(fileName));
      if (!codigo || !fmt) return;

      index[codigo + '|' + fmt + (isMissing ? '|FALTANTE' : '')] = true;

      if (!isMissing && fmt === 'json' && !codeMeta[codigo]) {
        codeMeta[codigo] = {
          classif: classificationFromLabel(config.name, label),
          fecha: formatYmd(parseEmissionDate(row[10])),
          tipoNombre: (row[15] || '').toString()
        };
      }
    });
  }

  config.dedupeIndex = index;
  config.codeMeta = codeMeta;
  return index;
}

function classificationFromLabel(companyName, label) {
  if (companyName === CONFIG.FOLDER_NO_PROCESADO) return noProcesadoClassification();
  var company = COMPANIES.find(function(c) { return c.name === companyName; }) || { name: companyName };
  if (label.endsWith(CONFIG.LABEL_VENTAS_SUFFIX)) return makeClassification(company, 'Ventas');
  if (label.endsWith(CONFIG.LABEL_COMPRAS_SUFFIX)) return makeClassification(company, 'Compras');
  return noProcesadoClassification();
}

function rememberCodeMeta(ctx, meta) {
  if (meta.codigo && meta.classif.type !== 'DTE') {
    ctx.threadCodeMeta[meta.codigo] = { classif: meta.classif, fecha: meta.fecha, tipoNombre: meta.tipoNombre };
  }
}

function lookupCodeMeta(ctx, codigo) {
  if (ctx.threadCodeMeta[codigo]) return ctx.threadCodeMeta[codigo];
  for (var i = 0; i < ctx.configs.length; i++) {
    getDedupeIndex(ctx.configs[i]);
    if (ctx.configs[i].codeMeta[codigo]) return ctx.configs[i].codeMeta[codigo];
  }
  return null;
}

function isAlreadyRegistered(ctx, codigo, fmt) {
  if (!codigo) return false;
  var key = codigo + '|' + fmt.toLowerCase();
  return ctx.configs.some(function(c) { return !!getDedupeIndex(c)[key]; });
}

function getOrCreateFolder(parent, name) {
  var iter = parent.getFoldersByName(name);
  return iter.hasNext() ? iter.next() : parent.createFolder(name);
}

function getOrCreateRootFolder() {
  if (CONFIG.DTE_ROOT_FOLDER_ID && CONFIG.DTE_ROOT_FOLDER_ID.trim() !== '') {
    try {
      var rootFolder = DriveApp.getFolderById(CONFIG.DTE_ROOT_FOLDER_ID);
      if (!rootFolder.isTrashed()) return rootFolder;
    } catch (e) {}
    Logger.log('⚠️ ID de carpeta inaccesible o en papelera. Buscando por nombre...');
  }

  var existingFolders = DriveApp.getFoldersByName(CONFIG.DTE_ROOT_FOLDER_NAME);
  while (existingFolders.hasNext()) {
    var folder = existingFolders.next();
    if (!folder.isTrashed()) return folder;
  }
  return DriveApp.createFolder(CONFIG.DTE_ROOT_FOLDER_NAME);
}

function getExt(name) { return (name || '').split('.').pop().toLowerCase(); }
function baseName(name) { var n = cleanFileName(name); var dot = n.lastIndexOf('.'); return dot > 0 ? n.substring(0, dot) : n; }
function cleanFileName(name) { return (name || '').toString().split(/[\\\/]/).pop(); }
function looksLikeDteName(name) { return GUID_REGEX.test(name || ''); }

/** JSON.parse tolerante: quita caracteres de control y recorta basura al final. */
function parseJSONSafe(content) {
  if (!content) return null;
  var cleanContent = content.replace(/^﻿/, '').trim()
    .replace(/\t/g, " ").replace(/[\x00-\x08\x0B-\x0C\x0E-\x1F]/g, "");
  try {
    return JSON.parse(cleanContent);
  } catch (e) {
    var lastBrace = cleanContent.lastIndexOf('}');
    while (lastBrace !== -1) {
      try {
        return JSON.parse(cleanContent.substring(0, lastBrace + 1));
      } catch (err) {
        lastBrace = cleanContent.lastIndexOf('}', lastBrace - 1);
      }
    }
  }
  return null;
}

function writeDataToSheetImmediately(sheet, rows) {
  var retryDelay = 1000;
  for (var attempt = 1; attempt <= 3; attempt++) {
    try {
      var last = sheet.getLastRow();
      if (last + rows.length > sheet.getMaxRows()) sheet.insertRowsAfter(sheet.getMaxRows(), rows.length);
      sheet.getRange(last + 1, 1, rows.length, rows[0].length).setValues(rows);
      SpreadsheetApp.flush();
      return true;
    } catch (e) {
      if (attempt < 3) {
        Utilities.sleep(retryDelay);
        retryDelay *= 2;
      }
    }
  }
  return false;
}

function getLabelPath(shortName) {
  var name = shortName.replace(/\s+/g, ' ').trim();
  return CONFIG.LABEL_PARENT ? CONFIG.LABEL_PARENT + '/' + name : name;
}

function normalizeLabelName(name) {
  return (name || '').toString().replace(/\s+/g, ' ').trim().toLowerCase();
}

/** Nombres reales (normalizados) en Gmail de las etiquetas del script. */
function getOwnLabelNames(allLabels) {
  return Object.keys(allLabels).map(function(n) { return normalizeLabelName(allLabels[n].getName()); });
}

/** Forma de búsqueda de Gmail: "DTE/No procesado" -> "dte-no-procesado". */
function toGmailSearchLabel(path) {
  return path.toLowerCase().replace(/[\s\/]+/g, '-');
}

function buildGmailSearchQuery(allLabels) {
  var parts = [];
  if (CONFIG.GMAIL_SEARCH_SCOPE) parts.push(CONFIG.GMAIL_SEARCH_SCOPE);
  if (CONFIG.GMAIL_SEARCH_AFTER) parts.push('after:' + CONFIG.GMAIL_SEARCH_AFTER);
  Object.keys(allLabels).forEach(function(name) {
    parts.push('-label:' + toGmailSearchLabel(allLabels[name].getName()));
  });
  return parts.join(' ');
}

function threadHasOwnLabel(thread, allLabels) {
  var own = getOwnLabelNames(allLabels);
  return thread.getLabels().some(function(l) { return own.indexOf(normalizeLabelName(l.getName())) !== -1; });
}

/**
 * Etiqueta única del hilo, para revisión manual primero: No procesado si algún DTE no se asignó a una
 * empresa; Solo PDF si lo pendiente son PDF sin JSON; si no, la empresa/tipo con más archivos
 * (en empate, el primero). Otros si no hay DTE.
 */
function resolveThreadLabel(records) {
  if (records.length === 0) return CONFIG.LABEL_OTROS;
  var pending = records.filter(function(r) { return r.classType === 'DTE'; });
  if (pending.some(function(r) { return r.typeLabel !== CONFIG.LABEL_SOLO_PDF; })) return CONFIG.FOLDER_NO_PROCESADO;
  if (pending.length > 0) return CONFIG.LABEL_SOLO_PDF;
  var counts = {};
  var order = [];
  records.forEach(function(r) {
    if (r.isMissing) return;
    if (!counts[r.typeLabel]) {
      counts[r.typeLabel] = 0;
      order.push(r.typeLabel);
    }
    counts[r.typeLabel]++;
  });
  if (order.length === 0) return CONFIG.FOLDER_NO_PROCESADO;
  return order.reduce(function(best, label) { return counts[label] > counts[best] ? label : best; }, order[0]);
}

/** Deja una sola etiqueta del script en el hilo. Las etiquetas del cliente no se tocan. */
function applyThreadLabel(thread, allLabels, shortName) {
  var target = allLabels[shortName];
  if (!target) {
    Logger.log('>>> [ETIQUETA] No existe la etiqueta "' + getLabelPath(shortName) + '"');
    return false;
  }

  var targetName = normalizeLabelName(target.getName());
  var own = getOwnLabelNames(allLabels).concat((CONFIG.LEGACY_LABELS || []).map(normalizeLabelName));

  thread.getLabels().forEach(function(l) {
    var name = normalizeLabelName(l.getName());
    if (name !== targetName && own.indexOf(name) !== -1) thread.removeLabel(l);
  });

  thread.addLabel(target);
  return true;
}

/**
 * Busca cada etiqueta del script y la crea solo si no existe. La búsqueda ignora mayúsculas y
 * espacios repetidos (Gmail los unifica). Si alguna no se puede preparar se detiene la ejecución:
 * sin etiqueta los correos no quedan marcados y se revisarían de nuevo en cada ejecución.
 */
function setupLabels() {
  var existing = {};
  function loadExisting() {
    GmailApp.getUserLabels().forEach(function(l) { existing[normalizeLabelName(l.getName())] = l; });
  }

  function ensureLabel(path) {
    var key = normalizeLabelName(path);
    if (existing[key]) return existing[key];
    try {
      existing[key] = GmailApp.createLabel(path);
      Logger.log('>>> [ETIQUETA] Creada: ' + path);
    } catch (e) {
      // Puede existir con otra forma de escribirse: se vuelve a buscar antes de darla por perdida.
      loadExisting();
      existing[key] = existing[key] || GmailApp.getUserLabelByName(path);
      if (!existing[key]) Logger.log('>>> [ETIQUETA] No se pudo crear ni encontrar "' + path + '": ' + e.toString());
    }
    return existing[key];
  }

  loadExisting();
  if (CONFIG.LABEL_PARENT) ensureLabel(CONFIG.LABEL_PARENT);

  var labels = {};
  var missing = [];
  getClassificationLabelNames().forEach(function(name) {
    var label = ensureLabel(getLabelPath(name));
    if (label) labels[name] = label;
    else missing.push(getLabelPath(name));
  });

  if (missing.length > 0) {
    throw new Error('No se pudieron preparar las etiquetas de Gmail: ' + missing.join(', ') +
      '. Revisar los nombres en CONFIG o si existen etiquetas con nombres parecidos.');
  }
  return labels;
}

function setupCompanyConfigs(root, headers) {
  var list = COMPANIES.concat([{ name: CONFIG.FOLDER_NO_PROCESADO }]);
  return list.map(function(c) {
    var folder = getOrCreateFolder(root, c.name);
    var sName = c.name + ' - Registros DTE';
    var files = folder.getFilesByName(sName);
    var ss;
    if (files.hasNext()) {
      ss = SpreadsheetApp.openById(files.next().getId());
    } else {
      ss = SpreadsheetApp.create(sName);
      var file = DriveApp.getFileById(ss.getId());
      folder.addFile(file);
      DriveApp.getRootFolder().removeFile(file);
    }
    var sheet = ss.getSheets()[0];
    ensureHeaders(sheet, headers);
    return { name: c.name, folder: folder, sheet: sheet, sheetName: sName };
  });
}

function ensureHeaders(sheet, headers) {
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(headers);
    return;
  }
  if (sheet.getMaxColumns() < headers.length) {
    sheet.insertColumnsAfter(sheet.getMaxColumns(), headers.length - sheet.getMaxColumns());
  }
  var current = sheet.getRange(1, 1, 1, headers.length).getValues()[0];
  headers.forEach(function(h, i) {
    if (current[i] === '') sheet.getRange(1, i + 1).setValue(h);
  });
}

function setupErrorLog(rootFolder) {
  var files = rootFolder.getFilesByName(CONFIG.ERROR_LOG_NAME);
  var ss;
  if (files.hasNext()) {
    ss = SpreadsheetApp.openById(files.next().getId());
  } else {
    ss = SpreadsheetApp.create(CONFIG.ERROR_LOG_NAME);
    var file = DriveApp.getFileById(ss.getId());
    rootFolder.addFile(file);
    DriveApp.getRootFolder().removeFile(file);
    ss.getSheets()[0].appendRow(['Fecha', 'Tipo Error', 'URL / Archivo', 'Remitente', 'Asunto', 'Link Correo', 'Detalle Técnico']);
  }
  return ss.getSheets()[0];
}

/** source puede ser los datos del correo o un registro (ambos tienen from, subject y enlace). */
function logToErrorSheet(sheet, type, target, source, detail) {
  var link = source.permalink || source.threadPermalink || '';
  sheet.appendRow([new Date(), type, target, source.from || '', source.subject || '', link, detail || ""]);
}
